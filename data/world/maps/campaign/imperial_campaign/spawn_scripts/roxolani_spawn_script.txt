script
	declare_persistent_counter steppe_faction_05_spawned

	if	I_CompareCounter steppe_faction_05_spawned == 0
	&&	not FactionIsAlive steppe_faction_05

		; Spawn if player
		if	I_LocalFaction steppe_faction_05
		&&	I_TurnNumber > 15
		&&	SettlementName Vicus_Venedae
		&&	RandomPercent > 98
			return true
		end_if
		
		if	I_TurnNumber > 200
		&&	I_TurnNumber < 210
		&&	SettlementName Vicus_Venedae
		&&	RandomPercent > 50
			return true
		end_if

		if	I_TurnNumber > 210
		&&	not IsCapital
		&&	HasResource s_paganism
		&&	SettlementLoyaltyLevel = loyalty_revolting
		&&	RandomPercent > 95
			return true
		end_if
	end_if
	
	return false
end_script
