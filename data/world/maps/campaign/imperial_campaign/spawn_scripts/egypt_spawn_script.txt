script
	declare_persistent_counter rebel_egypt_spawned

	if	I_CompareCounter rebel_egypt_spawned == 0
	&&	not FactionIsAlive rebel_egypt

		; Spawn if player
		if	I_LocalFaction rebel_egypt
		&&	I_TurnNumber > 15
		&&	not IsCapital
		&&	HasResource egyptian
		&&	RandomPercent > 98
			return true
		end_if

		if	I_TurnNumber > 130
		&&	I_TurnNumber < 170
		&&	SettlementName Waset
		&&	RandomPercent > 50
			return true
		end_if
		
		if	I_TurnNumber > 170
		&&	not IsCapital
		&&	HasResource egyptian
		&&	SettlementLoyaltyLevel = loyalty_revolting
		&&	RandomPercent > 95
			return true
		end_if
	end_if

	return false
end_script
