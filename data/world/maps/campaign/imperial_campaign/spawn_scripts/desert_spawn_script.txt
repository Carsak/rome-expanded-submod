script
	declare_persistent_counter rebel_desert_spawned
	declare_persistent_counter mercenary_spawned

	if	I_CompareCounter rebel_desert_spawned == 0
	&&	not FactionIsAlive rebel_desert

		; Spawn if player
		if	I_LocalFaction rebel_desert
		&&	I_TurnNumber > 15
		&&	not IsCapital
		&&	HasResource berber
		&&	RandomPercent > 98
			return true
		end_if
		
		if	I_CompareCounter mercenary_spawned == 1
		&&	I_TurnNumber > 55
		&&	I_TurnNumber < 65
		&&	SettlementName Thapsus
		&&	RandomPercent > 50
			return true
		end_if
		
		if	I_TurnNumber > 65
		&&	not IsCapital
		&&	HasResource berber
		&&	SettlementLoyaltyLevel = loyalty_revolting
		&&	RandomPercent > 95
			return true
		end_if
	end_if

	return false
end_script
