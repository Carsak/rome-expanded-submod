script
	declare_persistent_counter rebel_parthia_spawned

	if	I_CompareCounter rebel_parthia_spawned == 0
	&&	not FactionIsAlive rebel_parthia

		; Spawn if player
		if	I_LocalFaction rebel_parthia
		&&	I_TurnNumber > 15
		&&	not IsCapital
		&&	SettlementName Arsakia
		&&	RandomPercent > 98
			return true
		end_if

		if	I_TurnNumber > 40
		&&	I_TurnNumber < 50
		&&	SettlementName Arsakia
		&&	RandomPercent > 50
			return true
		end_if

		if	I_TurnNumber > 50
		&&	not IsCapital
		&&	HasResource zoroastrian
		&&	SettlementLoyaltyLevel = loyalty_revolting
		&&	RandomPercent > 95
			return true
		end_if
	end_if
	
	return false
end_script
